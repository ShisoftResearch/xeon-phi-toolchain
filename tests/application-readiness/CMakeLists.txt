cmake_minimum_required(VERSION 3.20)
project(k1om_application_readiness LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "k1om")
  message(FATAL_ERROR "Configure with cmake/k1om-toolchain.cmake")
endif()
find_package(Threads REQUIRED)
add_library(kernels STATIC ../thread-scaling-kernels.c)
target_compile_options(kernels PRIVATE -O3 -ffp-contract=off)
add_executable(knc-application main.cpp unwind.cpp)
set_target_properties(knc-application PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(knc-application PRIVATE -O2 -ffp-contract=off)
target_link_libraries(knc-application PRIVATE kernels Threads::Threads)
